Tabelle: anni, prodotti, ricavi_prodotti, zone, ricavi_zone, prodotti_servizi (valori in miliardi di dollari)
Filtri: prodotti[prodotto] (pagina 2) e zone[zona_breve] (pagina 4)
Collegamenti: anni <- ricavi_prodotti / ricavi_zone / prodotti_servizi · prodotti <- ricavi_prodotti · zone <- ricavi_zone

Ricavi Apple   (tabella anni) =
    SUM ( anni[ricavi] )
    formato: 0.0" mld $"

Utile Netto Apple   (tabella anni) =
    SUM ( anni[utile_netto] )
    formato: 0.0" mld $"

Ricavi 2025   (tabella anni) =
    CALCULATE ( [Ricavi Apple], REMOVEFILTERS ( anni ), anni[anno] = 2025 )
    formato: 0.0" mld $"

Volte dal 2010   (tabella anni) =

    formato: 0.0"x"

Crescita Anno   (tabella anni) =
    VAR a = MAX ( anni[anno] ) VAR prima = CALCULATE ( [Ricavi Apple], REMOVEFILTERS ( anni ), anni[anno] = a - 1 ) RETURN IF ( HASONEVALUE ( anni[anno] ) && NOT ISBLANK ( prima ), DIVIDE ( [Ricavi Apple] - prima, prima ) )
    formato: +0.0%;-0.0%
